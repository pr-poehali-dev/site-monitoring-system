# Рекомендации по улучшению паттернов парсинга связей между документами

## Текущее состояние (backend/link-finder/index.py)

Текущая реализация использует следующие ключевые фразы:

```python
context_keywords = [
    r'утратившим\s+силу',
    r'признать\s+утратившим',
    r'внести\s+изменени[яе]',
    r'дополнить',
    r'изложить\s+в\s+новой\s+редакции',
    r'в\s+редакции\s+постановлени',
]
```

И три основных паттерна извлечения дат/номеров:

1. `от DATE года №NUM` (обратный порядок)
2. `постановление/распоряжение №NUM от DATE` (с типом документа)
3. `№NUM от DATE` (стандартный)

### 5. Улучшение контекстного окна

```python
def get_context_window(text, keyword_match, keyword_pattern):
    """
    Умное определение размера контекстного окна в зависимости от фразы
    """
    if 'редакции' in keyword_pattern or 'изменениями' in keyword_pattern:
        # Для списков изменений - большое окно
        start_pos = max(0, keyword_match.start() - 100)
        end_pos = min(len(text), keyword_match.end() + 2000)
    elif 'утратившим силу' in keyword_pattern:
        # Для отмены - среднее окно
        start_pos = max(0, keyword_match.start() - 200)
        end_pos = min(len(text), keyword_match.end() + 500)
    else:
        # По умолчанию - стандартное окно
        start_pos = max(0, keyword_match.start() - 200)
        end_pos = min(len(text), keyword_match.end() + 300)
    
    return text[start_pos:end_pos]
```

### 6. Фильтрация ложных срабатываний

```python
def is_valid_reference(number: str, date_str: str, context: str) -> bool:
    """
    Проверка валидности найденной ссылки
    """
    # Фильтр 1: Номер должен быть разумным (не более 6 цифр)
    if len(number) > 6:
        return False
    
    # Фильтр 2: Дата не должна быть в далеком будущем или прошлом
    try:
        date_obj = datetime.strptime(date_str, '%Y-%m-%d')
        if date_obj.year < 1990 or date_obj.year > datetime.now().year + 1:
            return False
    except:
        return False
    
    # Фильтр 3: Контекст не должен быть внутри кавычек (цитата)
    if context.count('«') % 2 == 1 or context.count('"') % 2 == 1:
        return False
    
    # Фильтр 4: Исключить технические номера (телефоны, счета и т.п.)
    exclude_keywords = ['телефон', 'счет', 'банк', 'инн', 'окпо', 'снилс']
    context_lower = context.lower()
    if any(kw in context_lower for kw in exclude_keywords):
        return False
    
    return True
```

### 7. Приоритизация паттернов

```python
def prioritize_patterns(references: list) -> list:
    """
    Сортировка найденных ссылок по приоритету/релевантности
    """
    def get_priority(ref):
        context = ref.get('context', '').lower()
        
        # Высший приоритет - явная отмена
        if any(kw in context for kw in ['утратившим силу', 'признать утратившим']):
            return 1
        
        # Средний приоритет - изменения
        if any(kw in context for kw in ['внести изменения', 'дополнить', 'изложить']):
            return 2
        
        # Низкий приоритет - ссылки
        if any(kw in context for kw in ['на основании', 'в соответствии с']):
            return 3
        
        # Остальные
        return 4
    
    return sorted(references, key=get_priority)
```

def parse_document_references_improved(text: str) -> list:
    '''
    Улучшенная версия парсинга ссылок на документы
    '''
    references = []
    
    # Шаг 1: Нормализация текста
    text = normalize_text(text)
    
    # Шаг 2: Поиск ключевых фраз с расширенным списком
    for keyword_pattern in EXTENDED_CONTEXT_KEYWORDS:
        for keyword_match in re.finditer(keyword_pattern, text, re.IGNORECASE):
            context_text = get_context_window(text, keyword_match, keyword_pattern)
            
            # Шаг 3: Применение всех паттернов извлечения
            extracted_refs = []
            
            # 3.1: Обратный порядок (от DATE №NUM)
            extracted_refs.extend(extract_reverse_pattern(context_text))
            
            # 3.2: Прямой порядок (№NUM от DATE)
            extracted_refs.extend(extract_direct_pattern(context_text))
            
            # 3.3: С типом документа
            extracted_refs.extend(extract_with_doc_type(context_text))
            
            # 3.4: Диапазоны
            extracted_refs.extend(extract_range_pattern(context_text))
            
            # 3.5: Списки
            extracted_refs.extend(extract_list_pattern(context_text))
            
            # 3.6: Месяц словами
            extracted_refs.extend(extract_month_word_pattern(context_text))
            
            # Шаг 4: Фильтрация
            for ref in extracted_refs:
                if is_valid_reference(ref['number'], ref['date'], context_text):
                    ref['context'] = context_text
                    ref['keyword'] = keyword_match.group(0)
                    references.append(ref)
    
    # Шаг 5: Специальная обработка блоков в скобках
    references.extend(extract_from_brackets(text))
    
    # Шаг 6: Удаление дубликатов
    unique_refs = deduplicate_references(references)
    
    # Шаг 7: Приоритизация
    prioritized_refs = prioritize_patterns(unique_refs)
    
    return prioritized_refs

## Метрики качества

После внедрения улучшений рекомендуется отслеживать:

1. **Точность (Precision)**: Доля правильных ссылок среди найденных
2. **Полнота (Recall)**: Доля найденных ссылок среди всех существующих
3. **F1-мера**: Гармоническое среднее точности и полноты

Целевые показатели:
- Precision ≥ 95% (мало ложных срабатываний)
- Recall ≥ 85% (находим большинство ссылок)
- F1 ≥ 90%
